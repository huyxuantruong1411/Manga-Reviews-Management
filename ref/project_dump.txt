from __future__ import annotations

import json
import logging
import re
from collections import defaultdict
from typing import Any
from urllib.parse import parse_qs, urlparse

import requests

from core.base_api import BaseMangaAPI
from extractors.comix import ComixRequestSigner, ComixSignerError


logger = logging.getLogger(__name__)


class ComixAPI(BaseMangaAPI):
    BASE_URL = "https://comix.to"
    API_BASE_URL = f"{BASE_URL}/api/v2"
    CHAPTERS_PER_PAGE = 100

    def __init__(self) -> None:
        self.session = requests.Session()
        self.headers = {
            "User-Agent": (
                "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
                "AppleWebKit/537.36 (KHTML, like Gecko) "
                "Chrome/136.0.0.0 Safari/537.36"
            ),
            "Referer": f"{self.BASE_URL}/",
            "Accept-Language": "en-US,en;q=0.9",
            "Content-Type": "application/json",
        }
        self.signer = ComixRequestSigner()

    def search_manga(self, query: str) -> list[dict[str, Any]]:
        if query.startswith("http") and "comix.to/title/" in query:
            manga = self.get_manga_info(query)
            return [manga] if manga else []
        return []

    def get_manga_info(self, url: str) -> dict[str, Any] | None:
        title_url = self._normalize_title_url(url)
        if not title_url:
            return None

        response = self.session.get(title_url, headers=self.headers, timeout=20)
        response.raise_for_status()
        manga_payload = self._extract_manga_payload(response.text)
        if not manga_payload:
            return None

        preferred_group_id = self._extract_group_preference(title_url)
        manga = {
            "id": manga_payload.get("manga_id"),
            "hash_id": manga_payload.get("hash_id"),
            "title": manga_payload.get("title") or "Unknown",
            "alt_titles": manga_payload.get("alt_titles", []),
            "cover_url": (manga_payload.get("poster") or {}).get("large"),
            "author": ", ".join(author["title"] for author in manga_payload.get("author", [])) or "Unknown",
            "artist": ", ".join(artist["title"] for artist in manga_payload.get("artist", [])) or "Unknown",
            "rating": manga_payload.get("rated_avg") or 0,
            "follows": manga_payload.get("follows_total") or 0,
            "year": manga_payload.get("year") or "N/A",
            "preferred_group_id": preferred_group_id,
            "source_url": title_url,
        }
        manga["groups"] = self._get_groups(manga["hash_id"])
        chapter_items = self._load_all_chapter_items(manga)
        chapter_variants = self._group_chapter_variants(chapter_items)
        manga["chapter_variants"] = chapter_variants
        manga["chapters"] = self._build_download_path(chapter_variants, preferred_group_id)
        return manga

    def get_chapters(self, manga_id: str) -> list[dict[str, Any]]:
        info = self.get_manga_info(manga_id)
        return info["chapters"] if info else []

    def get_images(self, chapter_identifier: str | int) -> list[str]:
        return self.get_chapter_images(chapter_identifier)

    def get_chapter_images(self, chapter_identifier: str | int) -> list[str]:
        chapter_id = self._coerce_chapter_id(chapter_identifier)
        payload = self._request_json(f"/chapters/{chapter_id}")
        if not payload:
            return []
        return [image["url"] for image in payload.get("images", []) if image.get("url")]

    def get_chapter_detail(self, chapter_identifier: str | int) -> dict[str, Any] | None:
        chapter_id = self._coerce_chapter_id(chapter_identifier)
        return self._request_json(f"/chapters/{chapter_id}")

    def download_image(self, url: str, path: str) -> bool:
        try:
            with self.session.get(url, stream=True, timeout=20, headers=self.headers) as response:
                response.raise_for_status()
                with open(path, "wb") as file_obj:
                    for chunk in response.iter_content(chunk_size=8192):
                        if chunk:
                            file_obj.write(chunk)
            return True
        except Exception:
            return False

    def _load_all_chapter_items(self, manga: dict[str, Any]) -> list[dict[str, Any]]:
        items: list[dict[str, Any]] = []
        page = 1
        last_page = 1
        while page <= last_page:
            result = self._signed_request(
                manga,
                "/manga/{hash_id}/chapters".format(hash_id=manga["hash_id"]),
                {
                    "limit": self.CHAPTERS_PER_PAGE,
                    "page": page,
                    "order": {"number": "asc"},
                },
            )
            if not result:
                break
            items.extend(result.get("items", []))
            pagination = result.get("pagination", {})
            last_page = pagination.get("last_page", last_page)
            page += 1
        return items

    def _group_chapter_variants(self, items: list[dict[str, Any]]) -> list[dict[str, Any]]:
        grouped: dict[str, dict[str, Any]] = {}
        for item in items:
            chapter_number = item.get("number")
            chapter_key = self._chapter_key(chapter_number)
            group = item.get("scanlation_group") or {
                "scanlation_group_id": item.get("scanlation_group_id", 0),
                "name": "No Group",
            }
            option = {
                "id": item["chapter_id"],
                "chapter": self._format_chapter_number(chapter_number),
                "chapter_number": chapter_number,
                "title": item.get("name") or "",
                "volume": item.get("volume") or 0,
                "group_id": group.get("scanlation_group_id", 0),
                "group_name": group.get("name") or "No Group",
                "updated_at": item.get("updated_at") or 0,
                "votes": item.get("votes") or 0,
            }
            if chapter_key not in grouped:
                grouped[chapter_key] = {
                    "chapter": self._format_chapter_number(chapter_number),
                    "chapter_number": chapter_number,
                    "options": [],
                }
            grouped[chapter_key]["options"].append(option)

        for entry in grouped.values():
            entry["options"].sort(
                key=lambda option: (
                    option["group_name"] == "No Group",
                    -option["updated_at"],
                    -option["votes"],
                )
            )

        return sorted(grouped.values(), key=lambda entry: self._chapter_sort_value(entry["chapter_number"]))

    def _build_download_path(
        self,
        chapter_variants: list[dict[str, Any]],
        preferred_group_id: int | None,
    ) -> list[dict[str, Any]]:
        if not chapter_variants:
            return []

        availability: dict[int, set[str]] = defaultdict(set)
        chapter_lookup: list[dict[int, dict[str, Any]]] = []
        for entry in chapter_variants:
            options_by_group = {option["group_id"]: option for option in entry["options"]}
            chapter_lookup.append(options_by_group)
            for group_id in options_by_group:
                availability[group_id].add(self._chapter_key(entry["chapter_number"]))

        selected_group_id = self._resolve_initial_group(chapter_variants, availability, preferred_group_id)
        selected_chapters: list[dict[str, Any]] = []

        for index, entry in enumerate(chapter_variants):
            options_by_group = chapter_lookup[index]
            if selected_group_id not in options_by_group:
                selected_group_id = self._pick_fallback_group(
                    chapter_variants,
                    chapter_lookup,
                    index,
                    preferred_group_id,
                )
            chosen = dict(options_by_group[selected_group_id])
            chosen["switch_from_previous_group"] = (
                bool(selected_chapters) and selected_chapters[-1]["group_id"] != chosen["group_id"]
            )
            selected_chapters.append(chosen)

        return selected_chapters

    def _resolve_initial_group(
        self,
        chapter_variants: list[dict[str, Any]],
        availability: dict[int, set[str]],
        preferred_group_id: int | None,
    ) -> int:
        first_entry = chapter_variants[0]
        first_groups = {option["group_id"] for option in first_entry["options"]}
        if preferred_group_id in first_groups:
            return preferred_group_id

        candidates = list(first_groups)
        candidates.sort(
            key=lambda group_id: (
                -len(availability[group_id]),
                group_id != 0,
                group_id,
            )
        )
        return candidates[0]

    def _pick_fallback_group(
        self,
        chapter_variants: list[dict[str, Any]],
        chapter_lookup: list[dict[int, dict[str, Any]]],
        start_index: int,
        preferred_group_id: int | None,
    ) -> int:
        option_groups = list(chapter_lookup[start_index].keys())

        def score(group_id: int) -> tuple[int, int, int, int]:
            consecutive = 0
            remaining = 0
            for future_index in range(start_index, len(chapter_variants)):
                future_options = chapter_lookup[future_index]
                if group_id in future_options:
                    remaining += 1
                    if consecutive == future_index - start_index:
                        consecutive += 1
                elif consecutive == future_index - start_index:
                    break
            return (
                consecutive,
                remaining,
                1 if preferred_group_id == group_id else 0,
                1 if group_id == 0 else 0,
            )

        option_groups.sort(key=score, reverse=True)
        return option_groups[0]

    def _signed_request(
        self,
        manga: dict[str, Any],
        endpoint: str,
        params: dict[str, Any],
    ) -> dict[str, Any] | None:
        try:
            query = self.signer.sign_query(endpoint, params)
        except ComixSignerError as exc:
            logger.error("Could not sign Comix request for %s: %s", endpoint, exc)
            return None

        return self._request_json(endpoint, params=query, referer=manga["source_url"])

    def _get_groups(self, hash_id: str) -> list[dict[str, Any]]:
        payload = self._request_json(f"/manga/{hash_id}/groups")
        return payload or []

    def _request_json(
        self,
        endpoint: str,
        params: dict[str, Any] | None = None,
        referer: str | None = None,
    ) -> dict[str, Any] | list[Any] | None:
        headers = dict(self.headers)
        if referer:
            headers["Referer"] = referer
        response = self.session.get(
            f"{self.API_BASE_URL}{endpoint}",
            headers=headers,
            params=self._flatten_query_params(params),
            timeout=20,
        )
        response.raise_for_status()
        payload = response.json()
        if payload.get("status") == 200:
            return payload.get("result")
        logger.warning("Comix API returned %s for %s: %s", payload.get("status"), endpoint, payload.get("message"))
        return None

    @classmethod
    def _flatten_query_params(cls, params: dict[str, Any] | None) -> list[tuple[str, Any]] | None:
        if not params:
            return None

        pairs: list[tuple[str, Any]] = []

        def walk(prefix: str, value: Any) -> None:
            if value is None:
                return
            if isinstance(value, dict):
                for key, nested_value in value.items():
                    walk(f"{prefix}[{key}]", nested_value)
            elif isinstance(value, list):
                for item in value:
                    if isinstance(item, (dict, list)):
                        walk(f"{prefix}[]", item)
                    else:
                        pairs.append((f"{prefix}[]", item))
            else:
                pairs.append((prefix, value))

        for key, value in params.items():
            walk(key, value)

        return pairs

    @staticmethod
    def _extract_manga_payload(html: str) -> dict[str, Any] | None:
        match = re.search(r'\\"manga\\":(\{.*?\}),\\"scanlationGroups\\":', html, re.DOTALL)
        if not match:
            return None
        raw_object = match.group(1)
        try:
            return json.loads(json.loads('"' + raw_object + '"'))
        except json.JSONDecodeError:
            return None

    @staticmethod
    def _normalize_title_url(url: str) -> str | None:
        if not url:
            return None
        parsed = urlparse(url if url.startswith("http") else f"https://{url.lstrip('/')}" )
        if parsed.netloc not in {"comix.to", "www.comix.to"}:
            return None
        if not parsed.path.startswith("/title/"):
            return None
        normalized = parsed._replace(fragment="")
        return normalized.geturl().rstrip("/")

    @staticmethod
    def _extract_group_preference(url: str) -> int | None:
        query = parse_qs(urlparse(url).query)
        value = query.get("group", [None])[0]
        if value is None:
            return None
        try:
            return int(value)
        except (TypeError, ValueError):
            return None

    @staticmethod
    def _coerce_chapter_id(chapter_identifier: str | int) -> int:
        text = str(chapter_identifier)
        if text.isdigit():
            return int(text)
        prefix = text.split("-", 1)[0]
        if prefix.isdigit():
            return int(prefix)
        raise ValueError(f"Unsupported chapter identifier: {chapter_identifier}")

    @staticmethod
    def _chapter_key(number: Any) -> str:
        return str(number)

    @staticmethod
    def _chapter_sort_value(number: Any) -> tuple[float, str]:
        try:
            return (float(number), str(number))
        except (TypeError, ValueError):
            return (float("inf"), str(number))

    @staticmethod
    def _format_chapter_number(number: Any) -> str:
        try:
            numeric = float(number)
        except (TypeError, ValueError):
            return str(number)
        if numeric.is_integer():
            return str(int(numeric))
        return f"{numeric:g}"

import os
import json

OUTPUT_FILE = "project_dump.txt"

def should_ignore_dir(dirname: str) -> bool:
    """Bỏ qua thư mục venv (dù tên kiểu gì)."""
    return dirname.lower().startswith(("venv", 
                                       ".venv", 
                                       "__pycache__", 
                                       "Data", 
                                       "data", 
                                       ".git", 
                                       "flags", 
                                       ".gitignore", 
                                       ".gitattributes", 
                                       "pending-feature",
                                       "simulation",
                                       "debug",
                                       ".csv",
                                       ".log",
                                       "output",
                                       "project_dump.txt"))

def build_tree_dict(root_dir: str) -> dict:
    """Trả về cây thư mục dưới dạng dict (chỉ bỏ qua venv)."""
    tree = {"name": os.path.basename(root_dir), "type": "directory", "children": []}

    if should_ignore_dir(os.path.basename(root_dir)):
        return tree

    try:
        with os.scandir(root_dir) as it:
            for entry in sorted(it, key=lambda e: e.name):
                if entry.is_dir():
                    if should_ignore_dir(entry.name):
                        continue
                    tree["children"].append(build_tree_dict(entry.path))
                elif entry.is_file():
                    tree["children"].append({"name": entry.name, "type": "file"})
    except PermissionError:
        pass
    return tree

def build_tree_ascii(root_dir: str, prefix: str = "") -> str:
    """Tạo cây thư mục dạng text giống `tree` (chỉ bỏ qua venv)."""
    entries = []
    try:
        with os.scandir(root_dir) as it:
            for entry in sorted(it, key=lambda e: e.name):
                if entry.is_dir():
                    if should_ignore_dir(entry.name):
                        continue
                    entries.append(entry)
                elif entry.is_file():
                    entries.append(entry)
    except PermissionError:
        return ""

    lines = []
    for i, entry in enumerate(entries):
        connector = "└── " if i == len(entries) - 1 else "├── "
        if entry.is_dir():
            lines.append(prefix + connector + entry.name + "/")
            extension = "    " if i == len(entries) - 1 else "│   "
            lines.extend(build_tree_ascii(entry.path, prefix + extension).splitlines())
        else:
            lines.append(prefix + connector + entry.name)
    return "\n".join(lines)

def dump_project(root_dir: str, output_file: str):
    with open(output_file, "w", encoding="utf-8") as out:
        # Xuất cây thư mục dạng JSON
        tree_dict = build_tree_dict(root_dir)
        out.write("=== Project Tree (JSON) ===\n")
        out.write(json.dumps(tree_dict, indent=2, ensure_ascii=False))
        out.write("\n\n")

        # Xuất cây thư mục dạng ASCII
        out.write("=== Project Tree (ASCII) ===\n")
        out.write(root_dir + "/\n")
        out.write(build_tree_ascii(root_dir))
        out.write("\n\n")

        # Xuất nội dung tất cả file
        for dirpath, dirnames, filenames in os.walk(root_dir):
            # Lọc thư mục
            dirnames[:] = [d for d in dirnames if not should_ignore_dir(d)]

            rel_path = os.path.relpath(dirpath, root_dir)
            if rel_path == ".":
                rel_path = ""
            out.write(f"\n=== Folder: {rel_path or root_dir} ===\n")

            for filename in filenames:
                file_path = os.path.join(dirpath, filename)
                out.write(f"\n--- File: {file_path} ---\n")
                try:
                    with open(file_path, "r", encoding="utf-8") as f:
                        out.write(f.read())
                except Exception as e:
                    out.write(f"[Lỗi đọc file: {e}]\n")

if __name__ == "__main__":
    current_dir = os.getcwd()
    dump_project(current_dir, OUTPUT_FILE)
    print(f"✅ Đã xuất toàn bộ file (trừ venv) + cây thư mục JSON/ASCII vào {OUTPUT_FILE}")

import requests
import re
import logging

logging.basicConfig(level=logging.INFO, format='%(asctime)s - %(levelname)s - %(message)s')
logger = logging.getLogger(__name__)

class HentaiFoxAPI:
    HEADERS = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36",
        "Referer": "https://hentaifox.com/",
        "Accept-Language": "en-US,en;q=0.9"
    }

    def __init__(self):
        self.session = requests.Session()

    def get_gallery_info(self, gallery_id):
        """Lấy thông tin Title, Cover và Tổng số trang từ trang 1 của Gallery"""
        url = f"https://hentaifox.com/g/{gallery_id}/1/"
        try:
            resp = self.session.get(url, headers=self.HEADERS, timeout=15)
            if resp.status_code == 404 or "404 - Not Found" in resp.text:
                return None
            
            html = resp.text
            
            # 1. Trích xuất Title (Loại bỏ chữ " - Page 1 - HentaiFox")
            title_match = re.search(r'<title>(.*?)\s*-\s*Page 1', html)

def get_file_hash(filepath):
    hasher = hashlib.md5()
    try:
        with open(filepath, 'rb') as f:
            buf = f.read(131072) 
            while len(buf) > 0:
                hasher.update(buf)
                buf = f.read(131072)
        return hasher.hexdigest()
    except:
        return None

def is_uuid_filename(filename):
    name, _ = os.path.splitext(filename)
    pattern = r'^([a-f0-9]{32}|[a-f0-9]{8}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{12})$'
    return bool(re.match(pattern, name, re.IGNORECASE))

def format_bytes(size):
    power = 2**10
    n = 0
    power_labels = {0 : '', 1: 'K', 2: 'M', 3: 'G', 4: 'T'}
    while size > power:
        size /= power
        n += 1
    return f"{size:.2f} {power_labels[n]}B"

def convert_image_format(filepath, target_ext):
    """Hàm chuyển đổi định dạng ảnh sử dụng PIL"""
    if target_ext.lower() == "original": return filepath
    
    ext = target_ext.lower()
    if ext == ".jpg": ext_format = "JPEG"
    elif ext == ".png": ext_format = "PNG"
    elif ext == ".webp": ext_format = "WEBP"
    else: return filepath

    try:
        old_ext = os.path.splitext(filepath)[1].lower()
        if old_ext == ext:
            return filepath 

        new_path = os.path.splitext(filepath)[0] + ext
        
        with Image.open(filepath) as img:
            if ext_format == "JPEG" and img.mode in ("RGBA", "P"):
                img = img.convert("RGB")
            img.save(new_path, format=ext_format)
            
        if os.path.exists(filepath) and new_path != filepath:
            os.remove(filepath)
        return new_path
    except Exception as e:
        return filepath

    def start_scan_format(self):

    def _scan_format_thread(self, root_dir):
        temp_groups = {}
        total_checked = 0
        for dirpath, _, filenames in os.walk(root_dir):
            non_png_files = []
            for f in filenames:
                ext = os.path.splitext(f)[1].lower()
                if ext in ['.jpg', '.jpeg', '.bmp', '.webp']:
                    total_checked += 1
                    non_png_files.append(os.path.join(dirpath, f))
            if non_png_files: temp_groups[dirpath] = non_png_files
        self.non_png_groups = temp_groups
        self.after(0, lambda: self._render_format_results())

    def _render_format_results(self):
        count_files = sum(len(v) for v in self.non_png_groups.values())
        self.lbl_format_status.configure(text=f"Found {count_files} non-png images.")
        self.btn_scan_format.configure(state="normal", text="SCAN NON-PNG FILES")
        if count_files > 0: self.btn_convert_all_format.configure(state="normal")
        else: return ctk.CTkLabel(self.scroll_format, text="All files are already .png!", font=("Roboto", 16)).pack(pady=50)
        
        for folder, files in self.non_png_groups.items():
            GenericToolRow(self.scroll_format, folder, files, self.handle_format_action, action_name="PNG", btn_color=COLOR_PURPLE).pack(fill="x", pady=5, padx=5)

    def handle_format_action(self, action, target):
        files_to_convert = []
        if action == "convert_chapter" and target in self.non_png_groups:
            files_to_convert.extend(self.non_png_groups[target])
        elif action == "convert_manga":
            manga_dir = os.path.dirname(target)
            for folder, files in self.non_png_groups.items():
                if folder.startswith(manga_dir): files_to_convert.extend(files)
        elif action == "convert_all":
            if not messagebox.askyesno("Confirm", "Convert ALL found files to .png?"): return
            for files in self.non_png_groups.values(): files_to_convert.extend(files)

        if files_to_convert: threading.Thread(target=self._convert_files_to_png, args=(files_to_convert,), daemon=True).start()

    def _convert_files_to_png(self, file_list):
        total = len(file_list)
        self.after(0, lambda: self.btn_convert_all_format.configure(state="disabled", text="CONVERTING..."))
        
        converted_count = 0

        # Hàm task chạy độc lập trên từng luồng
        def convert_task(filepath):
            try:
                new_path = convert_image_format(filepath, ".png")
                return new_path != filepath # Trả về True nếu convert thành công (đường dẫn thay đổi)
            except:
                return False

        # Khởi tạo Pool với 8 luồng (workers)
        with concurrent.futures.ThreadPoolExecutor(max_workers=8) as executor:
            future_to_path = {executor.submit(convert_task, p): p for p in file_list}
            
            for future in concurrent.futures.as_completed(future_to_path):
                try:
                    success = future.result()
                    if success:
                        converted_count += 1
                        # Update UI an toàn qua main thread mỗi khi xong 5 file hoặc hoàn tất
                        if converted_count % 5 == 0 or converted_count == total:
                            self.after(0, lambda c=converted_count, t=total: self.lbl_format_status.configure(text=f"Converted: {c}/{t}"))
                except Exception:
                    pass

        self.after(0, lambda: self.lbl_format_status.configure(text="Format Conversion Finished."))
        self.after(0, lambda: self.btn_convert_all_format.configure(state="normal", text="CONVERT EVERYTHING TO .PNG"))
        self.after(0, lambda c=converted_count: messagebox.showinfo("Done", f"Converted {c} files to .PNG!"))
        self.after(0, self.start_scan_format)

    def _convert_files_to_uuid(self, file_list):

        self.after(0, lambda: self.btn_convert_all_uuid.configure(state="disabled", text="CONVERTING..."))
        
        converted_count = 0

        # Hàm task chạy độc lập
        def rename_task(old_path):
            if os.path.exists(old_path):
                try:
                    dirname, ext = os.path.dirname(old_path), os.path.splitext(old_path)[1]
                    os.rename(old_path, os.path.join(dirname, f"{uuid.uuid4().hex}{ext}"))
                    return True
                except:
                    pass
            return False

        # Khởi tạo Pool đa luồng
        with concurrent.futures.ThreadPoolExecutor(max_workers=8) as executor:
            future_to_path = {executor.submit(rename_task, p): p for p in file_list}
            
            for future in concurrent.futures.as_completed(future_to_path):
                try:
                    if future.result():
                        converted_count += 1
                        if converted_count % 10 == 0 or converted_count == total:
                            self.after(0, lambda c=converted_count, t=total: self.lbl_uuid_status.configure(text=f"Converted: {c}/{t}"))
                except Exception:
                    pass

        self.after(0, lambda: self.lbl_uuid_status.configure(text="Conversion Finished."))
        self.after(0, lambda: self.btn_convert_all_uuid.configure(state="normal", text="CONVERT EVERYTHING TO UUID"))

    def start_scan_duplicates(self):
        output_dir = os.path.join(os.path.dirname(os.path.abspath(__file__)), "output")
        if not os.path.exists(output_dir): return
        self.btn_scan.configure(state="disabled", text="SCANNING...")

    def _scan_process(self, root_dir):
        image_hashes, all_files = {}, []
        for dirpath, _, filenames in os.walk(root_dir):
            for f in filenames:
                if f.lower().endswith(('.png', '.jpg', '.jpeg', '.webp')): all_files.append(os.path.join(dirpath, f))
        
        with concurrent.futures.ThreadPoolExecutor(max_workers=8) as executor:
            future_to_path = {executor.submit(get_file_hash, p): p for p in all_files}
            for future in concurrent.futures.as_completed(future_to_path):
                fpath = future_to_path[future]
                try:
                    h = future.result()
                    if h: image_hashes.setdefault(h, []).append(fpath)
                except: pass
        self.duplicate_groups = {k: v for k, v in image_hashes.items() if len(v) > 1}
        self.after(0, self._render_duplicates)

    def _render_duplicates(self):
        count_groups = len(self.duplicate_groups)
        self.lbl_scan_status.configure(text=f"Found {count_groups} dupes.")
        self.btn_scan.configure(state="normal", text="SCAN 'output'")
        if count_groups == 0: return ctk.CTkLabel(self.scroll_cleaner, text="No duplicates found!", font=("Roboto", 16)).pack(pady=50)
        for h_val, paths in self.duplicate_groups.items():
            DuplicateGroupRow(self.scroll_cleaner, h_val, paths, self.handle_duplicate_action).pack(fill="x", pady=5, padx=5)

    def handle_duplicate_action(self, action, hash_val):
        if hash_val not in self.duplicate_groups: return
        paths = self.duplicate_groups[hash_val]
        files_to_delete = paths[1:] if action == "keep_one" else paths
        for p in files_to_delete:
            try: os.remove(p)
            except: pass
        del self.duplicate_groups[hash_val]
        for w in self.scroll_cleaner.winfo_children():
            if isinstance(w, DuplicateGroupRow) and w.hash_val == hash_val: w.destroy(); break

    def apply_all_duplicates(self, action):
        if not self.duplicate_groups or not messagebox.askyesno("Confirm", "Confirm Action?"): return
        for h in list(self.duplicate_groups.keys()): self.handle_duplicate_action(action, h)
        self.start_scan_duplicates()

        total_pages, completed, target_fmt = gal['total_pages'], 0, self.opt_format.get()
        base_img_url = gal['base_img_url']
        
        def download_page_task(p):
            ext = ".jpg"
            file_name = f"{uuid.uuid4().hex}{ext}" if self.switch_uuid.get() == 1 else f"{p:03d}{ext}"
            full_path = os.path.join(save_dir, file_name)

            img_url_jpg = f"{base_img_url}/{p}.jpg"
            success = self.threehentai_api.download_image(img_url_jpg, full_path)

                # Tự động Fallback sang định dạng .png nếu .jpg lỗi 404

                success = self.threehentai_api.download_image(img_url_png, full_path)

                img_url_webp = f"{base_img_url}/{p}.webp"

            if success and target_fmt != "Original":
                full_path = convert_image_format(full_path, target_fmt)
                
            return (p, full_path, success)

        with concurrent.futures.ThreadPoolExecutor(max_workers=4) as executor:
            future_to_page = {executor.submit(download_page_task, i): i for i in range(1, total_pages + 1)}

                        self.after(0, lambda fp=res[1]: self.update_preview_image(fp))

    def sort_results(self, choice):
        if not self.search_results or self.current_source in ["HentaiFox", "HentaiNexus", "TruyenQQ", "Comix", "3Hentai"]: return
        sl = self.search_results.copy()
        if choice == "Rating": sl.sort(key=lambda x: x['rating'], reverse=True)
        elif choice == "Follows": sl.sort(key=lambda x: x['follows'], reverse=True)
        elif choice == "Year": sl.sort(key=lambda x: str(x['year']) if x['year'] else "0", reverse=True)
        self.render_results(sl, self.entry_search.get().strip())

    def get_manga_info(self, url: str) -> dict[str, Any] | None:

            "preferred_group_id": preferred_group_id,
            "source_url": title_url,
        }
        manga["groups"] = self._get_groups(manga["hash_id"])

    def _load_all_chapter_items(self, manga: dict[str, Any]) -> list[dict[str, Any]]:

                    "limit": self.CHAPTERS_PER_PAGE,
                    "page": page,
                    "order": {"number": "asc"},

            if not result:
                break
            items.extend(result.get("items", []))
            pagination = result.get("pagination", {})
            last_page = pagination.get("last_page", last_page)
            page += 1
        return items

    def _group_chapter_variants(self, items: list[dict[str, Any]]) -> list[dict[str, Any]]:
        grouped: dict[str, dict[str, Any]] = {}
        for item in items:
            chapter_number = item.get("number")
            chapter_key = self._chapter_key(chapter_number)
            group = item.get("scanlation_group") or {
                "scanlation_group_id": item.get("scanlation_group_id", 0),
                "name": "No Group",
            }
            option = {
                "id": item["chapter_id"],
                "chapter": self._format_chapter_number(chapter_number),
                "chapter_number": chapter_number,
                "title": item.get("name") or "",
                "volume": item.get("volume") or 0,
                "group_id": group.get("scanlation_group_id", 0),
                "group_name": group.get("name") or "No Group",
                "updated_at": item.get("updated_at") or 0,
                "votes": item.get("votes") or 0,
            }
            if chapter_key not in grouped:
                grouped[chapter_key] = {
                    "chapter": self._format_chapter_number(chapter_number),
                    "chapter_number": chapter_number,
                    "options": [],
                }
            grouped[chapter_key]["options"].append(option)

        for entry in grouped.values():
            entry["options"].sort(
                key=lambda option: (
                    option["group_name"] == "No Group",
                    -option["updated_at"],
                    -option["votes"],
                )
            )

        return sorted(grouped.values(), key=lambda entry: self._chapter_sort_value(entry["chapter_number"]))

    def _build_download_path(
        self,
        chapter_variants: list[dict[str, Any]],
        preferred_group_id: int | None,
    ) -> list[dict[str, Any]]:
        if not chapter_variants:
            return []

        availability: dict[int, set[str]] = defaultdict(set)
        chapter_lookup: list[dict[int, dict[str, Any]]] = []
        for entry in chapter_variants:
            options_by_group = {option["group_id"]: option for option in entry["options"]}
            chapter_lookup.append(options_by_group)
            for group_id in options_by_group:
                availability[group_id].add(self._chapter_key(entry["chapter_number"]))

        selected_group_id = self._resolve_initial_group(chapter_variants, availability, preferred_group_id)
        selected_chapters: list[dict[str, Any]] = []

        for index, entry in enumerate(chapter_variants):
            options_by_group = chapter_lookup[index]
            if selected_group_id not in options_by_group:
                selected_group_id = self._pick_fallback_group(
                    chapter_variants,
                    chapter_lookup,
                    index,
                    preferred_group_id,
                )
            chosen = dict(options_by_group[selected_group_id])
            chosen["switch_from_previous_group"] = (
                bool(selected_chapters) and selected_chapters[-1]["group_id"] != chosen["group_id"]
            )
            selected_chapters.append(chosen)

        return selected_chapters

    def _resolve_initial_group(
        self,
        chapter_variants: list[dict[str, Any]],
        availability: dict[int, set[str]],
        preferred_group_id: int | None,
    ) -> int:
        first_entry = chapter_variants[0]
        first_groups = {option["group_id"] for option in first_entry["options"]}
        if preferred_group_id in first_groups:
            return preferred_group_id

        candidates = list(first_groups)
        candidates.sort(
            key=lambda group_id: (
                -len(availability[group_id]),
                group_id != 0,
                group_id,
            )
        )
        return candidates[0]

    def _pick_fallback_group(
        self,
        chapter_variants: list[dict[str, Any]],
        chapter_lookup: list[dict[int, dict[str, Any]]],
        start_index: int,
        preferred_group_id: int | None,
    ) -> int:
        option_groups = list(chapter_lookup[start_index].keys())

        def score(group_id: int) -> tuple[int, int, int, int]:
            consecutive = 0
            remaining = 0
            for future_index in range(start_index, len(chapter_variants)):
                future_options = chapter_lookup[future_index]
                if group_id in future_options:
                    remaining += 1
                    if consecutive == future_index - start_index:
                        consecutive += 1
                elif consecutive == future_index - start_index:
                    break
            return (
                consecutive,
                remaining,
                1 if preferred_group_id == group_id else 0,
                1 if group_id == 0 else 0,
            )

        option_groups.sort(key=score, reverse=True)
        return option_groups[0]

    def _signed_request(
        self,
        manga: dict[str, Any],
        endpoint: str,
        params: dict[str, Any],
    ) -> dict[str, Any] | None:
        try:
            query = self.signer.sign_query(endpoint, params)
        except ComixSignerError as exc:
            logger.error("Could not sign Comix request for %s: %s", endpoint, exc)
            return None

        return self._request_json(endpoint, params=query, referer=manga["source_url"])

    def _get_groups(self, hash_id: str) -> list[dict[str, Any]]:
        payload = self._request_json(f"/manga/{hash_id}/groups")
        return payload or []

    def _request_json(
        self,
        endpoint: str,
        params: dict[str, Any] | None = None,
        referer: str | None = None,
    ) -> dict[str, Any] | list[Any] | None:
        headers = dict(self.headers)
        if referer:
            headers["Referer"] = referer
        response = self.session.get(
            f"{self.API_BASE_URL}{endpoint}",
            headers=headers,
            params=self._flatten_query_params(params),
            timeout=20,
        )
        response.raise_for_status()
        payload = response.json()
        if payload.get("status") == 200:
            return payload.get("result")
        logger.warning("Comix API returned %s for %s: %s", payload.get("status"), endpoint, payload.get("message"))
        return None

    @classmethod
    def _flatten_query_params(cls, params: dict[str, Any] | None) -> list[tuple[str, Any]] | None:
        if not params:
            return None

        pairs: list[tuple[str, Any]] = []

        def walk(prefix: str, value: Any) -> None:
            if value is None:
                return
            if isinstance(value, dict):
                for key, nested_value in value.items():
                    walk(f"{prefix}[{key}]", nested_value)
            elif isinstance(value, list):
                for item in value:
                    if isinstance(item, (dict, list)):
                        walk(f"{prefix}[]", item)
                    else:
                        pairs.append((f"{prefix}[]", item))
            else:
                pairs.append((prefix, value))

        for key, value in params.items():
            walk(key, value)

        return pairs

def build_tree_dict(root_dir: str) -> dict:
    """Trả về cây thư mục dưới dạng dict (chỉ bỏ qua venv)."""
    tree = {"name": os.path.basename(root_dir), "type": "directory", "children": []}

    if should_ignore_dir(os.path.basename(root_dir)):
        return tree

    try:
        with os.scandir(root_dir) as it:
            for entry in sorted(it, key=lambda e: e.name):
                if entry.is_dir():
                    if should_ignore_dir(entry.name):
                        continue
                    tree["children"].append(build_tree_dict(entry.path))
                elif entry.is_file():
                    tree["children"].append({"name": entry.name, "type": "file"})
    except PermissionError:
        pass
    return tree

def build_tree_ascii(root_dir: str, prefix: str = "") -> str:
    """Tạo cây thư mục dạng text giống `tree` (chỉ bỏ qua venv)."""
    entries = []
    try:
        with os.scandir(root_dir) as it:
            for entry in sorted(it, key=lambda e: e.name):
                if entry.is_dir():
                    if should_ignore_dir(entry.name):
                        continue
                    entries.append(entry)
                elif entry.is_file():
                    entries.append(entry)
    except PermissionError:
        return ""

    lines = []
    for i, entry in enumerate(entries):
        connector = "└── " if i == len(entries) - 1 else "├── "
        if entry.is_dir():
            lines.append(prefix + connector + entry.name + "/")
            extension = "    " if i == len(entries) - 1 else "│   "
            lines.extend(build_tree_ascii(entry.path, prefix + extension).splitlines())
        else:
            lines.append(prefix + connector + entry.name)
    return "\n".join(lines)

def dump_project(root_dir: str, output_file: str):
    with open(output_file, "w", encoding="utf-8") as out:
        # Xuất cây thư mục dạng JSON
        tree_dict = build_tree_dict(root_dir)
        out.write("=== Project Tree (JSON) ===\n")
        out.write(json.dumps(tree_dict, indent=2, ensure_ascii=False))
        out.write("\n\n")

        # Xuất cây thư mục dạng ASCII
        out.write("=== Project Tree (ASCII) ===\n")
        out.write(root_dir + "/\n")
        out.write(build_tree_ascii(root_dir))
        out.write("\n\n")

        # Xuất nội dung tất cả file
        for dirpath, dirnames, filenames in os.walk(root_dir):
            # Lọc thư mục
            dirnames[:] = [d for d in dirnames if not should_ignore_dir(d)]

            rel_path = os.path.relpath(dirpath, root_dir)
            if rel_path == ".":
                rel_path = ""
            out.write(f"\n=== Folder: {rel_path or root_dir} ===\n")

            for filename in filenames:
                file_path = os.path.join(dirpath, filename)
                out.write(f"\n--- File: {file_path} ---\n")
                try:
                    with open(file_path, "r", encoding="utf-8") as f:
                        out.write(f.read())
                except Exception as e:
                    out.write(f"[Lỗi đọc file: {e}]\n")

def get_file_hash(filepath):
    hasher = hashlib.md5()
    try:
        with open(filepath, 'rb') as f:
            buf = f.read(131072) 
            while len(buf) > 0:
                hasher.update(buf)
                buf = f.read(131072)
        return hasher.hexdigest()
    except:
        return None

def format_bytes(size):
    power = 2**10
    n = 0
    power_labels = {0 : '', 1: 'K', 2: 'M', 3: 'G', 4: 'T'}
    while size > power:
        size /= power
        n += 1
    return f"{size:.2f} {power_labels[n]}B"

def convert_image_format(filepath, target_ext):
    """Hàm chuyển đổi định dạng ảnh sử dụng PIL"""
    if target_ext.lower() == "original": return filepath
    
    ext = target_ext.lower()
    if ext == ".jpg": ext_format = "JPEG"
    elif ext == ".png": ext_format = "PNG"
    elif ext == ".webp": ext_format = "WEBP"
    else: return filepath

    try:
        old_ext = os.path.splitext(filepath)[1].lower()
        if old_ext == ext:
            return filepath 

        new_path = os.path.splitext(filepath)[0] + ext
        
        with Image.open(filepath) as img:
            if ext_format == "JPEG" and img.mode in ("RGBA", "P"):
                img = img.convert("RGB")
            img.save(new_path, format=ext_format)
            
        if os.path.exists(filepath) and new_path != filepath:
            os.remove(filepath)
        return new_path
    except Exception as e:
        return filepath

    def start_scan_format(self):
        output_dir = os.path.join(os.path.dirname(os.path.abspath(__file__)), "output")
        if not os.path.exists(output_dir): return messagebox.showwarning("Warning", "Output folder not found!")
        self.btn_scan_format.configure(state="disabled", text="SCANNING...")
        self.btn_convert_all_format.configure(state="disabled")
        self.lbl_format_status.configure(text="Scanning file formats...")
        for widget in self.scroll_format.winfo_children(): widget.destroy()
        threading.Thread(target=self._scan_format_thread, args=(output_dir,), daemon=True).start()

    def _scan_format_thread(self, root_dir):
        temp_groups = {}
        total_checked = 0
        for dirpath, _, filenames in os.walk(root_dir):
            non_png_files = []
            for f in filenames:
                ext = os.path.splitext(f)[1].lower()
                if ext in ['.jpg', '.jpeg', '.bmp', '.webp']:
                    total_checked += 1
                    non_png_files.append(os.path.join(dirpath, f))
            if non_png_files: temp_groups[dirpath] = non_png_files
        self.non_png_groups = temp_groups
        self.after(0, lambda: self._render_format_results())

    def _render_format_results(self):
        count_files = sum(len(v) for v in self.non_png_groups.values())
        self.lbl_format_status.configure(text=f"Found {count_files} non-png images.")
        self.btn_scan_format.configure(state="normal", text="SCAN NON-PNG FILES")
        if count_files > 0: self.btn_convert_all_format.configure(state="normal")
        else: return ctk.CTkLabel(self.scroll_format, text="All files are already .png!", font=("Roboto", 16)).pack(pady=50)
        
        for folder, files in self.non_png_groups.items():
            GenericToolRow(self.scroll_format, folder, files, self.handle_format_action, action_name="PNG", btn_color=COLOR_PURPLE).pack(fill="x", pady=5, padx=5)

    def handle_format_action(self, action, target):
        files_to_convert = []
        if action == "convert_chapter" and target in self.non_png_groups:
            files_to_convert.extend(self.non_png_groups[target])
        elif action == "convert_manga":
            manga_dir = os.path.dirname(target)
            for folder, files in self.non_png_groups.items():
                if folder.startswith(manga_dir): files_to_convert.extend(files)
        elif action == "convert_all":
            if not messagebox.askyesno("Confirm", "Convert ALL found files to .png?"): return
            for files in self.non_png_groups.values(): files_to_convert.extend(files)

        if files_to_convert: threading.Thread(target=self._convert_files_to_png, args=(files_to_convert,), daemon=True).start()

    def _convert_files_to_png(self, file_list):
        total = len(file_list)
        self.after(0, lambda: self.btn_convert_all_format.configure(state="disabled", text="CONVERTING..."))
        
        converted_count = 0

        # Hàm task chạy độc lập trên từng luồng
        def convert_task(filepath):
            try:
                new_path = convert_image_format(filepath, ".png")
                return new_path != filepath # Trả về True nếu convert thành công (đường dẫn thay đổi)
            except:
                return False

        # Khởi tạo Pool với 8 luồng (workers)
        with concurrent.futures.ThreadPoolExecutor(max_workers=8) as executor:
            future_to_path = {executor.submit(convert_task, p): p for p in file_list}
            
            for future in concurrent.futures.as_completed(future_to_path):
                try:
                    success = future.result()
                    if success:
                        converted_count += 1
                        # Update UI an toàn qua main thread mỗi khi xong 5 file hoặc hoàn tất
                        if converted_count % 5 == 0 or converted_count == total:
                            self.after(0, lambda c=converted_count, t=total: self.lbl_format_status.configure(text=f"Converted: {c}/{t}"))
                except Exception:
                    pass

        self.after(0, lambda: self.lbl_format_status.configure(text="Format Conversion Finished."))
        self.after(0, lambda: self.btn_convert_all_format.configure(state="normal", text="CONVERT EVERYTHING TO .PNG"))
        self.after(0, lambda c=converted_count: messagebox.showinfo("Done", f"Converted {c} files to .PNG!"))
        self.after(0, self.start_scan_format)

    def _scan_uuid_thread(self, root_dir):
        temp_groups = {}
        for dirpath, _, filenames in os.walk(root_dir):
            non_uuid_files = [os.path.join(dirpath, f) for f in filenames if f.lower().endswith(('.png', '.jpg', '.jpeg', '.webp')) and not is_uuid_filename(f)]
            if non_uuid_files: temp_groups[dirpath] = non_uuid_files
        self.non_uuid_groups = temp_groups
        self.after(0, lambda: self._render_uuid_results())

    def _render_uuid_results(self):
        count_files = sum(len(v) for v in self.non_uuid_groups.values())

        for folder, files in self.non_uuid_groups.items():
            GenericToolRow(self.scroll_uuid, folder, files, self.handle_uuid_action).pack(fill="x", pady=5, padx=5)

    def handle_uuid_action(self, action, target):
        files_to_convert = []
        if action == "convert_chapter" and target in self.non_uuid_groups: files_to_convert.extend(self.non_uuid_groups[target])
        elif action == "convert_manga":
            manga_dir = os.path.dirname(target)
            for folder, files in self.non_uuid_groups.items():
                if folder.startswith(manga_dir): files_to_convert.extend(files)
        elif action == "convert_all":
            if not messagebox.askyesno("Confirm", "Convert ALL to UUID?"): return
            for files in self.non_uuid_groups.values(): files_to_convert.extend(files)
        if files_to_convert: threading.Thread(target=self._convert_files_to_uuid, args=(files_to_convert,), daemon=True).start()

    def _convert_files_to_uuid(self, file_list):
        total = len(file_list)
        self.after(0, lambda: self.btn_convert_all_uuid.configure(state="disabled", text="CONVERTING..."))
        
        converted_count = 0

        # Hàm task chạy độc lập
        def rename_task(old_path):
            if os.path.exists(old_path):
                try:
                    dirname, ext = os.path.dirname(old_path), os.path.splitext(old_path)[1]
                    os.rename(old_path, os.path.join(dirname, f"{uuid.uuid4().hex}{ext}"))
                    return True
                except:
                    pass
            return False

        # Khởi tạo Pool đa luồng
        with concurrent.futures.ThreadPoolExecutor(max_workers=8) as executor:
            future_to_path = {executor.submit(rename_task, p): p for p in file_list}
            
            for future in concurrent.futures.as_completed(future_to_path):
                try:
                    if future.result():
                        converted_count += 1
                        if converted_count % 10 == 0 or converted_count == total:
                            self.after(0, lambda c=converted_count, t=total: self.lbl_uuid_status.configure(text=f"Converted: {c}/{t}"))
                except Exception:
                    pass

        self.after(0, lambda: self.lbl_uuid_status.configure(text="Conversion Finished."))
        self.after(0, lambda: self.btn_convert_all_uuid.configure(state="normal", text="CONVERT EVERYTHING TO UUID"))
        self.after(0, lambda c=converted_count: messagebox.showinfo("Done", f"Converted {c} files successfully!"))
        self.after(0, self.start_scan_uuid)

    def start_scan_duplicates(self):
        output_dir = os.path.join(os.path.dirname(os.path.abspath(__file__)), "output")
        if not os.path.exists(output_dir): return
        self.btn_scan.configure(state="disabled", text="SCANNING...")

        threading.Thread(target=self._scan_process, args=(output_dir,), daemon=True).start()

    def _scan_process(self, root_dir):
        image_hashes, all_files = {}, []
        for dirpath, _, filenames in os.walk(root_dir):
            for f in filenames:
                if f.lower().endswith(('.png', '.jpg', '.jpeg', '.webp')): all_files.append(os.path.join(dirpath, f))
        
        with concurrent.futures.ThreadPoolExecutor(max_workers=8) as executor:
            future_to_path = {executor.submit(get_file_hash, p): p for p in all_files}
            for future in concurrent.futures.as_completed(future_to_path):
                fpath = future_to_path[future]
                try:
                    h = future.result()
                    if h: image_hashes.setdefault(h, []).append(fpath)
                except: pass
        self.duplicate_groups = {k: v for k, v in image_hashes.items() if len(v) > 1}
        self.after(0, self._render_duplicates)

    def _render_duplicates(self):
        count_groups = len(self.duplicate_groups)
        self.lbl_scan_status.configure(text=f"Found {count_groups} dupes.")
        self.btn_scan.configure(state="normal", text="SCAN 'output'")
        if count_groups == 0: return ctk.CTkLabel(self.scroll_cleaner, text="No duplicates found!", font=("Roboto", 16)).pack(pady=50)
        for h_val, paths in self.duplicate_groups.items():
            DuplicateGroupRow(self.scroll_cleaner, h_val, paths, self.handle_duplicate_action).pack(fill="x", pady=5, padx=5)

    def handle_duplicate_action(self, action, hash_val):
        if hash_val not in self.duplicate_groups: return
        paths = self.duplicate_groups[hash_val]
        files_to_delete = paths[1:] if action == "keep_one" else paths
        for p in files_to_delete:
            try: os.remove(p)
            except: pass
        del self.duplicate_groups[hash_val]
        for w in self.scroll_cleaner.winfo_children():
            if isinstance(w, DuplicateGroupRow) and w.hash_val == hash_val: w.destroy(); break

    def apply_all_duplicates(self, action):
        if not self.duplicate_groups or not messagebox.askyesno("Confirm", "Confirm Action?"): return
        for h in list(self.duplicate_groups.keys()): self.handle_duplicate_action(action, h)
        self.start_scan_duplicates()

        total_pages, completed, target_fmt = gal['total_pages'], 0, self.opt_format.get()

        def download_page_task(p):
            ext = ".jpg"
            file_name = f"{uuid.uuid4().hex}{ext}" if self.switch_uuid.get() == 1 else f"{p:03d}{ext}"
            full_path = os.path.join(save_dir, file_name)

            if success and target_fmt != "Original":
                full_path = convert_image_format(full_path, target_fmt)
                
            return (p, full_path, success)

        with concurrent.futures.ThreadPoolExecutor(max_workers=4) as executor:
            future_to_page = {executor.submit(download_page_task, i): i for i in range(1, total_pages + 1)}
            for future in concurrent.futures.as_completed(future_to_page):

class MangaDexAPI:
    BASE_URL = "https://api.mangadex.org"
    
    # [UPDATE] Thay đổi User-Agent thành Chrome thật để tránh bị chặn bot
    HEADERS = {
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36",
        "Referer": "https://mangadex.org/",
        "Origin": "https://mangadex.org",
        "Accept-Language": "en-US,en;q=0.9",
        "Connection": "keep-alive"
    }

    # Danh sách nội dung (bao gồm 18+)
    CONTENT_RATINGS = ["safe", "suggestive", "erotica", "pornographic"]

    def __init__(self):
        self.session = requests.Session()
        # [UPDATE] Tăng khả năng retry khi gặp lỗi mạng
        retries = Retry(
            total=5, 
            backoff_factor=1.5, 
            status_forcelist=[429, 500, 502, 503, 504],
            allowed_methods=["HEAD", "GET", "OPTIONS"]
        )
        self.session.mount('https://', HTTPAdapter(max_retries=retries))

    def _request(self, endpoint, params=None):
        try:
            # [UPDATE] Thêm timeout dài hơn chút
            resp = self.session.get(
                f"{self.BASE_URL}{endpoint}", 
                params=params, 
                headers=self.HEADERS, 
                timeout=15
            )
            
            if resp.status_code == 429:
                retry_after = int(resp.headers.get('Retry-After', 5))
                logger.warning(f"Rate limited. Sleeping {retry_after}s...")
                time.sleep(retry_after)
                return self._request(endpoint, params)
            
            resp.raise_for_status()
            return resp.json()
        except requests.exceptions.ConnectionError as e:
            logger.error(f"Connection Error (ISP Block suspected?): {e}")
            return None
        except Exception as e:
            logger.error(f"API Error ({endpoint}): {e}")
            return None

    def search_manga_advanced(self, query):
        """Search trả về đầy đủ thông tin: Title, Author, Artist, Year, Cover URL, Alt Titles."""
        if self._is_uuid(query):
            params = {"ids[]": [query]}
        else:
            params = {
                "title": query, 
                "limit": 20, 
                "order[relevance]": "desc",
            }
        
        params["contentRating[]"] = self.CONTENT_RATINGS
        params["includes[]"] = ["author", "artist", "cover_art"]

        data = self._request("/manga", params)
        if not data or "data" not in data:
            return []

        manga_list = []
        manga_ids = []

        for item in data["data"]:
            attr = item["attributes"]
            m_id = item["id"]
            manga_ids.append(m_id)

            title = attr["title"].get("en") or list(attr["title"].values())[0] if attr["title"] else "No Title"
            
            # [MOI] Lay danh sach ten phu (Alt Titles)
            alt_titles = []
            if "altTitles" in attr:
                for alt_dict in attr["altTitles"]:
                    # alt_dict la cac dict nhu {"vi": "Ten TV"}, {"ja": "Ten Nhat"} -> lay values
                    alt_titles.extend(alt_dict.values())

            authors = [rel["attributes"]["name"] for rel in item["relationships"] if rel["type"] == "author" and "attributes" in rel]
            artists = [rel["attributes"]["name"] for rel in item["relationships"] if rel["type"] == "artist" and "attributes" in rel]
            
            cover_file = None
            for rel in item["relationships"]:
                if rel["type"] == "cover_art" and "attributes" in rel:
                    cover_file = rel["attributes"]["fileName"]
                    break
            
            cover_url = f"https://uploads.mangadex.org/covers/{m_id}/{cover_file}.256.jpg" if cover_file else None

            manga_list.append({
                "id": m_id,
                "title": title,
                "alt_titles": alt_titles, # [MOI] Them truong nay
                "year": attr.get("year") or "N/A",
                "author": ", ".join(authors) if authors else "Unknown",
                "artist": ", ".join(artists) if artists else "Unknown",
                "cover_url": cover_url,
                "status": attr.get("status"),
                "follows": 0,
                "rating": 0.0
            })

                m["follows"] = s.get("follows", 0)

                "offset": offset,

                    "title": attr.get("title") or "",

from io import BytesIO
from typing import Optional

import requests
from PIL import Image


THUMBNAIL_TIMEOUT = 5


def load_image_from_url(url: str, size: tuple[int, int] = (50, 70)) -> Optional[Image.Image]:
    """Download and resize an image for preview usage."""
    try:
        response = requests.get(url, timeout=THUMBNAIL_TIMEOUT)
        response.raise_for_status()
        image = Image.open(BytesIO(response.content))
        image.thumbnail(size, Image.Resampling.LANCZOS)
        return image
    except Exception:
        return None
